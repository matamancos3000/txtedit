import re
from pathlib import Path
import os
import csv

# ================== CONFIGURACIÓN ==================
CARPETA_RAIZ = Path(r"C:\Users\45056372\Desktop\prueba its\calidad2025")
CARPETA_DE_RECURSOS = "Formatos2021"
URL_SHAREPOINT = "https://teams.global.hsbc/sites/labpspyas-2019/CALIDAD20251/"
LOG_CSV = CARPETA_RAIZ.parent / "log_cambios.csv"


# ================== FUNCIONES ==================
def convertir_enlaces(contenido, ruta_html):
    """
    Convierte enlaces dentro de un HTML:
    - HTML (.htm/.html) -> PDF en SharePoint
    - Documentos -> carpeta de recursos
    - Soporta enlaces absolutos, relativos, ../
    - Soporta carpetas genéricas ../*/archivo.htm (dinámico)
    """
    cambios = []

    def reemplazar_enlace(match):
        comilla = match.group(1)
        enlace = match.group(2)
        enlace_norm = enlace.replace("\\", "/")

        print(f"\n      🔍 Analizando enlace: {enlace_norm}")

        # ---------------- Fragmento (#...) ----------------
        if "#" in enlace_norm:
            ruta_sin_fragmento, fragmento = enlace_norm.split("#", 1)
        else:
            ruta_sin_fragmento = enlace_norm
            fragmento = None

        nombre_base, extension = os.path.splitext(os.path.basename(ruta_sin_fragmento))
        extension = extension.lower()

        # ==================================================
        # CASO: HTML
        # ==================================================
        if extension in [".htm", ".html"]:

            # ----- CASO ESPECIAL: ../cualquier_carpeta/archivo.htm (DINÁMICO) -----
            # Patrón mejorado para capturar cualquier carpeta después de ../
            m = re.search(r'\.\./([^/]+)/(.+\.htm[l]?)$', ruta_sin_fragmento, flags=re.IGNORECASE)
            if m:
                carpeta_origen = m.group(1)  # Captura el nombre de la carpeta (ej: documentos_apoyo_uso_comun)
                archivo = m.group(2)         # Captura el nombre del archivo
                
                print(f"         ✓ Caso especial dinámico: ../{carpeta_origen}/")
                print(f"         📁 Carpeta origen detectada: {carpeta_origen}")
                print(f"         📄 Archivo: {archivo}")

                # Convertir a PDF y mantener la misma estructura de carpetas
                ruta_relativa = Path(carpeta_origen) / Path(archivo).with_suffix(".pdf")
                nuevo = f"{URL_SHAREPOINT}{ruta_relativa.as_posix()}"
                
                print(f"         ✓ Nueva URL: {nuevo}")
                cambios.append((enlace, nuevo, f"HTML->PDF (../{carpeta_origen})"))
                return f'href={comilla}{nuevo}{comilla}'

            # ----- Caso 1: ya contiene calidad2021/ -----
            m = re.search(r'calidad2021/', ruta_sin_fragmento, flags=re.IGNORECASE)
            if m:
                suffix = ruta_sin_fragmento[m.end():]
                ruta_relativa = Path(suffix).with_suffix(".pdf").as_posix()
                nuevo = f"{URL_SHAREPOINT}{ruta_relativa}"
                print(f"         ✓ Caso 1: contiene calidad2021/")
                cambios.append((enlace, nuevo, "HTML->PDF"))
                return f'href={comilla}{nuevo}{comilla}'

            # ----- Caso 2: empieza con / -----
            if ruta_sin_fragmento.startswith("/"):
                ruta_relativa = (
                    Path(ruta_sin_fragmento.lstrip("/"))
                    .with_suffix(".pdf")
                    .as_posix()
                )
                nuevo = f"{URL_SHAREPOINT}{ruta_relativa}"
                print(f"         ✓ Caso 2: empieza con /")
                cambios.append((enlace, nuevo, "HTML->PDF (raiz)"))
                return f'href={comilla}{nuevo}{comilla}'

            # ----- Caso 3: rutas relativas normales -----
            try:
                ruta_completa = (ruta_html.parent / ruta_sin_fragmento).resolve()
                print(f"         🔧 Caso 3: intentando resolver ruta...")
                print(f"         📍 Ruta completa: {ruta_completa}")
                
                try:
                    rel = ruta_completa.relative_to(CARPETA_RAIZ)
                    ruta_relativa = Path(rel).with_suffix(".pdf").as_posix()
                    nuevo = f"{URL_SHAREPOINT}{ruta_relativa}"

                    tipo = "HTML->PDF (con ../)" if ".." in enlace else "HTML->PDF (relativo)"
                    print(f"         ✓ Ruta resuelta: {rel}")
                    cambios.append((enlace, nuevo, tipo))
                    return f'href={comilla}{nuevo}{comilla}'
                except ValueError:
                    print(f"         ⚠️ Fuera de CARPETA_RAIZ")
                    return match.group(0)
            except Exception as e:
                print(f"         ⚠️ Error resolviendo: {e}")
                return match.group(0)

        # ==================================================
        # CASO: DOCUMENTOS
        # ==================================================
        elif extension in [".docx", ".pdf", ".xlsx", ".pptx", ".doc", ".xls", ".xlsm"]:
            m = re.search(r'calidad2021/formatos2021/', ruta_sin_fragmento, flags=re.IGNORECASE)
            if m:
                suffix = ruta_sin_fragmento[m.end():]
                nuevo = f"{URL_SHAREPOINT}{CARPETA_DE_RECURSOS}/{suffix}"
                print(f"         ✓ Documento convertido")
                cambios.append((enlace, nuevo, extension.upper().strip(".")))
                return f'href={comilla}{nuevo}{comilla}'
            else:
                print(f"         ⚠️ Documento sin calidad2021/formatos2021/")

        return match.group(0)

    patron = r'''
        href=
        (["\'])
        ([^"\']+\.(?:htm[l]?|docx|pdf|xlsx|pptx|doc|xls|xlsm)(?:\#[^"']+)?)
        \1
    '''

    nuevo_contenido = re.sub(
        patron,
        reemplazar_enlace,
        contenido,
        flags=re.IGNORECASE | re.VERBOSE
    )

    return nuevo_contenido, cambios


def procesar_html(html_path):
    try:
        contenido = html_path.read_text(encoding="utf-8")
        encoding = "utf-8"
        print(f"   📖 Leyendo con UTF-8")
    except UnicodeDecodeError:
        contenido = html_path.read_text(encoding="latin-1")
        encoding = "latin-1"
        print(f"   📖 Leyendo con Latin-1")

    contenido_modificado, cambios = convertir_enlaces(contenido, html_path)
    return contenido_modificado, cambios, encoding


def registrar_cambios(archivo_html, cambios):
    existe = LOG_CSV.exists()
    with open(LOG_CSV, "a", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        if not existe:
            writer.writerow(["Archivo HTML", "Tipo", "Enlace original", "Enlace nuevo"])
        for original, nuevo, tipo in cambios:
            writer.writerow([archivo_html, tipo, original, nuevo])


# ================== PROCESO PRINCIPAL ==================
print("=" * 60)
print("🚀 INICIANDO CONVERSIÓN DE ENLACES")
print("=" * 60)
print(f"📂 CARPETA_RAIZ: {CARPETA_RAIZ}")
print(f"🔗 URL_SHAREPOINT: {URL_SHAREPOINT}")
print(f"📝 LOG_CSV: {LOG_CSV}")
print("=" * 60)

procesados = set()
archivos_procesados = 0
total_modificados = 0

for html_file in CARPETA_RAIZ.rglob("*.[hH][tT][mM]*"):

    # Ignorar carpetas _archivos
    if any(part.endswith("_archivos") for part in html_file.parts):
        continue

    if html_file in procesados:
        continue

    procesados.add(html_file)
    archivos_procesados += 1

    print(f"\n📄 Procesando: {html_file.relative_to(CARPETA_RAIZ)}")
    print("-" * 40)

    try:
        contenido, cambios, encoding = procesar_html(html_file)

        if cambios:
            html_file.write_text(contenido, encoding=encoding)
            registrar_cambios(str(html_file.relative_to(CARPETA_RAIZ)), cambios)
            total_modificados += len(cambios)
            print(f"\n   ✅ {len(cambios)} enlace(s) modificados en este archivo")
            
            # Mostrar resumen de cambios
            for i, (orig, nuevo, tipo) in enumerate(cambios, 1):
                print(f"      {i}. {tipo}")
                print(f"         Original: {orig}")
                print(f"         Nuevo:    {nuevo}")
        else:
            print("   ⚠️ Sin cambios")

    except Exception as e:
        print(f"   ❌ Error: {e}")
        import traceback
        traceback.print_exc()

print("\n" + "=" * 60)
print("🏁 PROCESO COMPLETADO")
print("=" * 60)
print(f"📄 HTML procesados: {archivos_procesados}")
print(f"🔗 Enlaces modificados: {total_modificados}")
print(f"📝 Log CSV: {LOG_CSV}")
print("=" * 60)
